# gocv v0.31.0 (OpenCV 4.6.0) CUDA bindings against Iluvatar CoreX (clang++/ivcore11)
#
# Key finding (2026-07-21): the environment now ships a CoreX build of OpenCV 4.6.0
# under /usr/local/corex (headers in include/opencv4, libs incl. libopencv_cuda*.so).
# gocv HEAD (v0.43.0) targets OpenCV 4.12; the matching gocv release for OpenCV 4.6.0
# is v0.31.0, checked out as a git worktree at .tools/gocv-0.31.
#
# gocv itself does NOT compile CUDA kernels — cuda/ is a CGO wrapper over OpenCV's
# CUDA modules. So the migration reduces to building/linking Go+CGO against the
# already-CoreX-built OpenCV, no nvcc/clang kernel porting required here.
#
# Environment:
export PATH="$PWD/.tools/go/bin:$PATH"                      # Go 1.25.5 (user-local)
export PKG_CONFIG_PATH=/usr/local/corex/lib64/pkgconfig
export PKG_CONFIG="pkg-config --define-prefix"              # opencv4.pc hardcodes /usr/local; --define-prefix fixes it
export LD_LIBRARY_PATH=/usr/local/corex/lib64:/usr/local/corex/lib:$LD_LIBRARY_PATH
export CUDA_VISIBLE_DEVICES=0,1
unset CCC_OVERRIDE_OPTIONS                                  # no kernel compilation here; leave clang flags alone
#
# CGO flags:
export CGO_CFLAGS="-I/usr/local/corex/include/opencv4"
export CGO_CPPFLAGS="-I/usr/local/corex/include/opencv4 -D_GLIBCXX_USE_CXX11_ABI=0"
export CGO_CXXFLAGS="-I/usr/local/corex/include/opencv4 -D_GLIBCXX_USE_CXX11_ABI=0"
export CGO_LDFLAGS="-L/usr/local/corex/lib64"
#   -D_GLIBCXX_USE_CXX11_ABI=0 : CoreX OpenCV libs export the pre-C++11 std::string
#     ABI (nm -D shows RKSs / no __cxx11), so gocv wrappers must be compiled with the
#     old ABI or every cv:: string symbol is undefined at link time.
#
# Run:
cd .tools/gocv-0.31
go test -v -count=1 ./cuda
# Crashers isolated for a clean run (all are CoreX OpenCV lib gaps, not gocv bugs):
go test -v -count=1 -skip '^(TestFlip|TestHoughSegment|TestSparsePyrLKOpticalFlow)' ./cuda
